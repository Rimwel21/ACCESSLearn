```python
from sqlalchemy import create_engine
from sqlalchemy.orm import sessionmaker, declarative_base
from core.config import settings

# SQLite requires check_same_thread=False for FastAPI's async threading
connect_args = {"check_same_thread": False} if settings.database_url.startswith("sqlite") else {}

engine_kwargs = {"connect_args": connect_args}
if not settings.database_url.startswith("sqlite"):
    engine_kwargs.update({
        "pool_pre_ping": True,
        "pool_recycle": 1800,
    })

engine = create_engine(settings.database_url, **engine_kwargs)

SessionLocal = sessionmaker(
    autoflush=False,
    autocommit=False,
    bind=engine
)

Base = declarative_base()
```

This module configures the database layer. SIGNHEAR uses SQLite for local development and Supabase PostgreSQL for production or staging. PostgreSQL connections use pre-ping and connection recycling; the request-scoped `get_db()` dependency is defined in `utils/dependencies.py`.

3. ACCOUNT AUTHENTICATION
============================================================

File: E-Learn/Backend/auth/account_auth.py

```python
from datetime import datetime, timedelta, timezone
from jose import jwt
from passlib.context import CryptContext
from core.config import settings

pwd_context = CryptContext(schemes=["bcrypt"], deprecated="auto")

def hash_password(password: str):
    return pwd_context.hash(password)

def verify_password(plain: str, hashed: str) -> bool:
    return pwd_context.verify(plain, hashed)

def create_access_token(data: dict):
    to_encode = data.copy()
    expire = datetime.now(timezone.utc) + timedelta(
        minutes=settings.access_token_expire_minutes
    )
    to_encode.update({"exp": expire})
    return jwt.encode(to_encode, settings.secret_key, algorithm=settings.algorithm)
```

This file handles password hashing, password verification, and JWT access-token creation. It supports secure login sessions for student, teacher, and administrator accounts.

```python
ALLOWED_GRADE_NAMES = ("Grade 4", "Grade 5", "Grade 6")

def validate_supported_grade_name(name: str) -> str:
    normalized = name.strip()
    if normalized not in ALLOWED_GRADE_NAMES:
        raise ValueError("Only Grade 4, Grade 5, and Grade 6 are supported.")
    return normalized
```

This logic limits SIGNHEAR academic workflows to supported grade levels only. Grade 1 to Grade 3 records are hidden or blocked from system workflows, while Grade 4 to Grade 6 remain available.

5. TEACHER SECTION ROUTES
============================================================

File: E-Learn/Backend/routes/teacher_section_route.py

```python
from fastapi import APIRouter, Depends, HTTPException
from sqlalchemy.orm import Session
from database.connection import get_db
from utils.dependencies import get_current_user
from models.accounts import Accounts

router = APIRouter(prefix="/teacher/sections", tags=["Teacher Sections"])

@router.get("/available")
def available_sections(db: Session = Depends(get_db), current_user: Accounts = Depends(get_current_user)):
    if current_user.role != "teacher":
        raise HTTPException(status_code=403, detail="Teacher access required")
    return {"sections": []}

@router.post("/{section_id}/select")
def select_section(section_id: int, db: Session = Depends(get_db), current_user: Accounts = Depends(get_current_user)):
    if current_user.role != "teacher":
        raise HTTPException(status_code=403, detail="Teacher access required")
    return {"section_id": section_id, "selected": True}
```

These routes allow a teacher to view available sections and select a section for class handling. This supports teacher workflows for modules, activities, quizzes, and student records.

6. HANDSIGN ROUTES
============================================================

File: E-Learn/Backend/routes/handsign_route.py

```python
from fastapi import APIRouter, Depends, HTTPException, Request
from utils.dependencies import get_current_user
from models.accounts import Accounts
from schemas.handsign.prediction import PredictionRequest, PredictionResponse

router = APIRouter(prefix="/api/handsign", tags=["Hand Sign Language"])

@router.get("/health")
def health(current_user: Accounts = Depends(get_current_user)):
    return {"status": "ready"}

@router.post("/predict", response_model=PredictionResponse)
def predict(payload: PredictionRequest, request: Request, current_user: Accounts = Depends(get_current_user)):
    service = getattr(request.app.state, "handsign_prediction_service", None)
    if service is None:
        raise HTTPException(status_code=503, detail="Prediction service unavailable")
    return service.predict_frame(payload.image)
```

The hand-sign route group provides camera-based sign-language recognition endpoints. It includes health checking, image prediction, camera detection, tutorial videos, dataset labels, and practice scoring.

7. FRONTEND API WRAPPER
============================================================

File: E-Learn/frontend/src/lib/api.ts

```typescript
export const API_BASE_URL = import.meta.env.VITE_API_BASE_URL || "http://localhost:8000"

export async function apiFetch<T>(path: string, options: RequestInit = {}): Promise<T> {
  const token = localStorage.getItem("access_token")
  const response = await fetch(`${API_BASE_URL}${path}`, {
    ...options,
    headers: {
      "Content-Type": "application/json",
      ...(token ? { Authorization: `Bearer ${token}` } : {}),
      ...(options.headers || {}),
    },
  })

  if (!response.ok) {
    const message = await response.text()
    throw new Error(message || "Request failed")
  }

  return response.json()
}
```

This frontend helper centralizes API requests. It attaches the stored JWT token, sends requests to the backend, and raises readable errors when requests fail.

```typescript
import axios from "axios"
import { API_BASE_URL } from "@/lib/api"

const handsignApi = axios.create({ baseURL: `${API_BASE_URL}/api/handsign` })

handsignApi.interceptors.request.use((config) => {
  const token = localStorage.getItem("access_token")
  if (token) config.headers.Authorization = `Bearer ${token}`
  return config
})

export async function checkHandsignHealth() {
  const response = await handsignApi.get("/health")
  return response.data
}

export async function predictHandSign(image: string) {
  const response = await handsignApi.post("/predict", { image })
  return response.data
}
```

This service connects the frontend camera/sign-language components to the backend recognition API. It sends authenticated requests for health checks, predictions, tutorial scoring, and practice records.

12. OFFLINE LOGIN SUPPORT
============================================================

File: E-Learn/frontend/src/lib/offlineDb.ts

```typescript
export type OfflineAccount = {
  username: string
  role: "student" | "teacher" | "admin"
  passwordHash: string
  savedAt: string
}

const OFFLINE_ACCOUNTS_KEY = "signhear_offline_accounts"

export function saveOfflineAccount(account: OfflineAccount) {
  const records = getOfflineAccounts()
  const next = records.filter((item) => item.username !== account.username)
  next.push(account)
  localStorage.setItem(OFFLINE_ACCOUNTS_KEY, JSON.stringify(next))
}

export function getOfflineAccounts(): OfflineAccount[] {
  const raw = localStorage.getItem(OFFLINE_ACCOUNTS_KEY)
  return raw ? JSON.parse(raw) : []
}
```

Offline support allows a user who has logged in online before to access cached content on the same device when internet is unavailable. Registration still requires internet access.
